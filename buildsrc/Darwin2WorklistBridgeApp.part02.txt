          string stateText = startUtc.Ticks.ToString(CultureInfo.InvariantCulture) +
                                   "|" +
                                   effectiveNow.Ticks.ToString(CultureInfo.InvariantCulture);
                byte[] encrypted = ProtectedData.Protect(
                    Encoding.UTF8.GetBytes(stateText),
                    Entropy,
                    DataProtectionScope.CurrentUser);
                File.WriteAllBytes(StatePath, encrypted);

                return new TrialStatus
                {
                    IsValid = true,
                    IsExpired = expired,
                    RemainingDays = remaining,
                    StartUtc = startUtc,
                    ExpiresUtc = expiresUtc,
                    Error = string.Empty
                    ,MachineCode = machineCode
                };
            }
            catch (Exception ex)
            {
                return new TrialStatus
                {
                    IsValid = false,
                    IsExpired = true,
                    RemainingDays = 0,
                    Error = ex.Message
                };
            }
        }
    }

        private void StartBridge(bool showMessages)
        {
            if (trial == null || !trial.IsValid || trial.IsExpired || (trial.Features & Features.Worklist) == 0)
            {
                if (showMessages) MessageBox.Show("Worklist 功能未授權或試用已到期。", "授權", MessageBoxButtons.OK, MessageBoxIcon.Warning);
                return;
            }
            try
            {
                var scriptPath = Path.Combine(baseDirectory, "Start-DarwinWorklistBridge.ps1");
                if (!File.Exists(scriptPath))
                {
                    throw new FileNotFoundException("找不到 Start-DarwinWorklistBridge.ps1", scriptPath);
                }
                RunPowerShellScript(scriptPath, true);
                TryStartDicomClient();
                if (showMessages) MessageBox.Show("Darwin2 Worklist 橋接程式已啟動。", "完成", MessageBoxButtons.OK, MessageBoxIcon.Information);
            }
            catch (Exception ex)
            {
                if (showMessages) MessageBox.Show(ex.Message, "啟動失敗", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
            RefreshStatus();
        }

        private void StopBridge(bool showMessages)
        {
            try
            {
                DicomMwlSettingsForm.StopClient(baseDirectory);
                Process bridgeProcess = GetBridgeProcess(true);
                if (bridgeProcess != null)
                {
                    bridgeProcess.Kill();
                    bridgeProcess.WaitForExit(5000);
                }
                if (File.Exists(lockPath)) File.Delete(lockPath);
                if (showMessages) MessageBox.Show("Darwin2 Worklist 橋接程式已停止。", "完成", MessageBoxButtons.OK, MessageBoxIcon.Information);
            }
            catch (Exception ex)
            {
                if (showMessages) MessageBox.Show(ex.Message, "停止失敗", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
            RefreshStatus();
        }

        private void TryStartDicomClient()
        {
            string scriptPath = Path.Combine(baseDirectory, "Start-DicomMwlClient.ps1");
            if (!File.Exists(scriptPath)) return;

            string configPath = Path.Combine(baseDirectory, "config", "hospital-interface.json");
            if (!File.Exists(configPath)) return;
            try
            {
                var root = new JavaScriptSerializer().DeserializeObject(File.ReadAllText(configPath, Encoding.UTF8)) as Dictionary<string, object>;
                var dicom = root == null ? null : root["dicomMwl"] as Dictionary<string, object>;
                object enabledValue;
                bool enabled = dicom != null && dicom.TryGetValue("enabled", out enabledValue) && Convert.ToBoolean(enabledValue);
                if (enabled)
                {
                    DicomMwlSettingsForm.StartClient(baseDirectory);
                }
            }
            catch
            {
            }
        }

        private Process GetBridgeProcess(bool requireRunning)
        {
            foreach (Process process in Process.GetProcessesByName("powershell"))
            {
                try
                {
                    using (var searcher = new ManagementObjectSearcher("SELECT CommandLine FROM Win32_Process WHERE ProcessId=" + process.Id))
                    {
                        foreach (ManagementObject obj in searcher.Get())
                        {
                            string cmd = Convert.ToString(obj["CommandLine"]);
                            if (cmd.IndexOf("Start-DarwinWorklistBridge.ps1", StringComparison.OrdinalIgnoreCase) >= 0) return process;
                        }
                    }
                }
                catch { }
            }
            return null;
        }

        {
            bool running = GetBridgeProcess(true) != null;
            statusLabel.Text = running ? "● 執行中" : "● 已停止";

            if (DateTime.UtcNow >= nextTrialRefreshUtc)
            {
                trial = TrialManager.GetOrCreate();
                nextTrialRefreshUtc = DateTime.UtcNow.AddMinutes(10);
                UpdateTrialDisplay();
                ApplyFeatureControls();
            }
            RefreshLog();
        }

        private void RefreshLog()
        {
            try
            {
                if (!File.Exists(logPath)) { logTextBox.Text = "尚無紀錄。"; return; }
                string[] lines = File.ReadAllLines(logPath, Encoding.UTF8);
                logTextBox.Lines = lines.Skip(Math.Max(0, lines.Length - 80)).ToArray();
                logTextBox.SelectionStart = logTextBox.TextLength;
                logTextBox.ScrollToCaret();
            }
            catch { }
        }

                File.Copy(sourcePath, destination, true);

            {
                MessageBox.Show(ex.Message, "匯入失敗", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private void SearchWorklist(string tag, string patientId)

            {
                string exePath = Path.Combine(baseDirectory, "Darwin2DicomMwlManual.exe");
                if (!File.Exists(exePath)) throw new FileNotFoundException("找不到 Darwin2DicomMwlManual.exe", exePath);
                var arguments = new List<string> { "--query", "--tag", tag };
                if (!string.IsNullOrWhiteSpace(patientId)) { arguments.Add("--value"); arguments.Add(patientId); }
                var psi = new ProcessStartInfo(exePath, string.Join(" ", arguments.Select(QuoteArgument))) { WorkingDirectory = baseDirectory, UseShellExecute = false, CreateNoWindow = true, RedirectStandardOutput = true, RedirectStandardError = true };
                using (var process = Process.Start(psi))
                {
                    string output = process.StandardOutput.ReadToEnd();
                    string error = process.StandardError.ReadToEnd();
                    process.WaitForExit();
                    string text = string.IsNullOrWhiteSpace(output) ? error : output;
                    SearchResultForm.Display(this, text, patientId.Length == 0 ? "Worklist 更新結果" : "搜尋結果");
                }
            }
            catch (Exception ex)
            {
                MessageBox.Show(ex.Message, "Worklist 查詢失敗", MessageBoxButtons.OK, MessageBoxIcon.Error);
            }
        }

        private static string QuoteArgument(string value)
        {
            return "\"" + (value ?? string.Empty).Replace("\"", "\\\"") + "\"";
        }

        private static void OpenDirectory(string directory)
        {
            Directory.CreateDirectory(directory);
            Process.Start(new ProcessStartInfo("explorer.exe", "\"" + directory + "\"") { UseShellExecute = true });
        }

        private static void RunPowerShellScript(string scriptPath, bool hidden)
        {
            var psi = new ProcessStartInfo
            {
                FileName = "powershell.exe",
                Arguments = "-NoProfile -ExecutionPolicy Bypass -File \"" + scriptPath + "\"",
                WorkingDirectory = Path.GetDirectoryName(scriptPath),
                UseShellExecute = false,
                CreateNoWindow = hidden,
                WindowStyle = hidden ? ProcessWindowStyle.Hidden : ProcessWindowStyle.Normal
            };
            Process.Start(psi);
        }
    }
